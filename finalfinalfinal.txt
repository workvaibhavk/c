## Practical 3 --- Call by value

``` cpp
#include <iostream>


void change(int x) {
    x = 100;
    cout << "Inside function: " << x << '\n';
}

int main() {
    int a = 10;
    change(a);
    cout << "In main: " << a << '\n';
    return 0;
}
```

**Point:** The function receives a copy, so the original `a` remains
`10`.

## Practical 4 --- Call by reference

``` cpp
#include <iostream>


void swapNumbers(int &a, int &b) {
    int temp = a;
    a = b;
    b = temp;
}

int main() {
    int x = 10, y = 20;
    swapNumbers(x, y);
    cout << "x = " << x << "\ny = " << y;
    return 0;
}
```

**Point:** Reference parameters are aliases for the original variables;
their values change.

## Practical 5 --- Function overloading

``` cpp
#include <iostream>


int area(int side) {
    return side * side;
}

int area(int length, int breadth) {
    return length * breadth;
}

int main() {
    cout << "Area of square = " << area(5) << '\n';
    cout << "Area of rectangle = " << area(5, 3) << '\n';
    return 0;
}
```

**Point:** Same function name, different parameter lists. Return type
alone cannot overload a function.

## Practical 9 --- Static data member and static member function

``` cpp
#include <iostream>


class Student {

        cin >> roll;
        count++;
    }
```

## Practical 13 --- String operations using predefined functions

This version uses C-style character arrays and `<cstring>`.

``` cpp
#include <iostream>
#include <cstring>


int main() {
    char a[100] = "Hello";
    char b[100] = "World";
    char joined[210];

    strcpy(joined, a);
    strcat(joined, b);
    cout << "Concatenation: " << joined << '\n';
    cout << "Comparison (0 means equal): " << strcmp(a, b) << '\n';

    // Reverse using a loop; standard <cstring> has no portable strrev().
    int n = strlen(text);
    for (int i = n - 1; i >= 0; i--) cout << text[i];

```

**Note:** `strrev()` is compiler-specific and is not part of standard
C++. Reversing with a loop is portable.

## Practical 14 --- String operations without predefined string functions

``` cpp
#include <iostream>


int main() {
    char a[50] = "Hello";
    char b[50] = "World";
    char joined[100];
    int i = 0, j = 0;

    while (a[i] != '\0') {
        joined[i] = a[i];
        i++;
    }
    while (b[j] != '\0') {
        joined[i] = b[j];
        i++;
        j++;
    }
    joined[i] = '\0';
    cout << "Concatenation: " << joined << '\n';

    int k = 0;
    while (a[k] != '\0' && b[k] != '\0' && a[k] == b[k]) k++;
    if (a[k] == b[k])
        cout << "Strings are equal\n";
    else
        cout << "Strings are not equal\n";

    char text[] = "computer";
    char target = 'p';
    int position = -1;
    for (int x = 0; text[x] != '\0'; x++) {
        if (text[x] == target) {
            position = x;
            break;
        }
    }
    cout << "Position of p (starting from 0): " << position << '\n';

    int length = 0;
    while (text[length] != '\0') length++;
    cout << "Reversed: ";
    for (int x = length - 1; x >= 0; x--) cout << text[x];
    cout << '\n';
    return 0;
}
```

## Practical 18 --- Virtual base class (diamond problem)

``` cpp
#include <iostream>


class Person {
public:
    int id;
};

class Student : virtual public Person { };
class Employee : virtual public Person { };

class Intern : public Student, public Employee {
public:
    void display() {
        cout << "ID = " << id;
    }
};

int main() {
    Intern x;
    x.id = 101;
    x.display();
    return 0;
}
```

**Point:** `virtual public Person` ensures `Intern` contains one shared
`Person` base subobject rather than two copies.

## Practical 19 --- Unary and binary operator overloading

``` cpp
#include <iostream>


class Number {
    int value;
public:
    Number(int v = 0) : value(v) { }

    void operator-() {       // unary minus operator
        value = -value;
    }

    Number operator+(const Number &n) {  // binary plus operator
        return Number(value + n.value);
    }

    void display() const {
        cout << value << '\n';
    }
};

int main() {
    Number a(10), b(20);
    -a;
    cout << "After unary minus: ";
    a.display();

    Number c = a + b;
    cout << "After binary addition: ";
    c.display();
    return 0;
}
```

## Practical 20 --- Operator overloading using a friend function

``` cpp
#include <iostream>


class Number {
    int value;
public:
    Number(int v = 0) : value(v) { }

    friend Number operator+(const Number &, const Number &);
    friend Number operator-(const Number &);
    void display() const {
        cout << value << '\n';
    }
};

Number operator+(const Number &a, const Number &b) {
    return Number(a.value + b.value);
}

Number operator-(const Number &a) {
    return Number(-a.value);
}

int main() {
    Number a(10), b(4);
    Number c = a + b;
    Number d = -a;
    cout << "Binary addition: ";
    c.display();
    cout << "Unary minus: ";
    d.display();
    return 0;
}
```

**Point:** A friend operator function is not a member function, but it
can access private members.

## Practical 21 --- `this` pointer

``` cpp
#include <iostream>


class Number {
    int value;
public:
    void setValue(int value) {
        this->value = value;
    }
    void display() {
        cout << "Value = " << this->value;
    }
};

int main() {
    Number n;
    n.setValue(50);
    n.display();
    return 0;
}
```

**Point:** `this` points to the current object. `this->value` means the
object's data member, while `value` is the parameter.

## Practical 22 --- Virtual function

``` cpp
#include <iostream>

class Shape {
public:
    virtual void display() {
        cout << "This is a shape\n";
    }
    virtual ~Shape() { }
};

class Circle : public Shape {
public:
    void display() override {
        cout << "This is a circle\n";
    }
};

int main() {
    Circle c;
    Shape *ptr = &c;
    ptr->display();
    return 0;
}
```

**Point:** Because `display()` is virtual, the call through a base-class
pointer runs the derived-class version.

## Practical 23 --- Type conversion

This program demonstrates basic-to-class, class-to-basic, and
one-class-to-another conversion.

``` cpp
#include <iostream>


class Distance {
    float metres;
public:
    Distance(float m) : metres(m) { }  // basic type to class type

    operator float() const {           // class type to basic type
        return metres;
    }

    float getMetres() const {
        return metres;
    }
};

class Centimetres {
    float cm;
public:
    Centimetres(const Distance &d) {   // class type to another class type
        cm = d.getMetres() * 100;
    }

    void display() const {
        cout << cm << " cm\n";
    }
};

int main() {
    Distance d = 2.5f;       // basic -> class
    float metres = d;        // class -> basic
    Centimetres c = d;       // class -> class

    cout << "Metres = " << metres << '\n';
    cout << "Converted to centimetres: ";
    c.display();
    return 0;
}
```

## Practical 24A --- File processing

``` cpp
#include <iostream>
#include <fstream>


int main() {
    ofstream out("sample.txt");
    if (!out) {
        cout << "Unable to open file for writing";
        return 1;
    }
    out << "Object Oriented Programming\n";
    out << "File handling practical\n";
    out.close();

    if (!in) {
        cout << "Unable to open file for reading";

        cout << line << '\n';
```

**Demonstrates:** opening, writing, reading and closing a text file. The
file is created in the program's working directory.

## Practical 24B --- Exception handling

``` cpp
#include <iostream>


int main() {
    int a, b;
    cout << "Enter numerator and denominator: ";
    cin >> a >> b;

    try {
        if (b == 0) {
            throw b;
        }
        cout << "Result = " << static_cast<double>(a) / b;
    }
    catch (int) {
        cout << "Exception: division by zero is not allowed";
    }
    return 0;
}
```

**Demonstrates:** `try`, `throw` and `catch`. This handles the specific
division-by-zero case.
